function safeFormatReportDate(val) {
  if (!val) return '';
  if (val instanceof Date && !isNaN(val.getTime())) {
    return Utilities.formatDate(val, 'GMT+8', 'yyyy-MM-dd');
  }
  const str = String(val).trim();
  if (str.length >= 10 && str.charAt(4) === '-' && str.charAt(7) === '-') {
    return str.substring(0, 10);
  }
  return str;
}

/**
 * Retrieves the Notification Roster for all departments
 */
function getNotificationRoster() {
  try {
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    let sheet = ss.getSheetByName('Notification_Roster');

    if (!sheet) {
      sheet = ss.insertSheet('Notification_Roster');
      sheet.appendRow(['Department Category', 'Notification Emails (Comma Separated)', 'Active Status', 'Last Updated']);
      sheet.getRange('A1:D1').setBackground('#0F4C3A').setFontColor('#FFFFFF').setFontWeight('bold');
      sheet.setFrozenRows(1);

      const defaultRows = [
        ['COMPLIANCE AND RISK', Session.getActiveUser().getEmail() || 'compliance@cardmri.com', 'Active', new Date()],
        ['ACCOUNTING', Session.getActiveUser().getEmail() || 'accounting@cardmri.com', 'Active', new Date()],
        ['CLAIMS', 'claims@cardmri.com', 'Active', new Date()],
        ['UNDERWRITING', 'underwriting@cardmri.com', 'Active', new Date()],
        ['IT', 'it.risk@cardmri.com', 'Active', new Date()],
        ['DCC', 'dcc@cardmri.com', 'Active', new Date()],
        ['PERSONNEL FILE', 'hr@cardmri.com', 'Active', new Date()],
        ['BOAT', 'boat@cardmri.com', 'Active', new Date()],
        ['MBA COORDINATOR', 'coordinator@cardmri.com', 'Active', new Date()]
      ];
      sheet.getRange(2, 1, defaultRows.length, 4).setValues(defaultRows);
    }

    const rows = sheet.getLastRow() > 1 ? sheet.getRange(2, 1, sheet.getLastRow() - 1, 4).getValues() : [];
    const roster = {};

    rows.forEach(r => {
      const dept = String(r[0] || '').trim().toUpperCase();
      const emails = String(r[1] || '').trim();
      const status = String(r[2] || 'Active').trim();
      if (dept) {
        roster[dept] = {
          emails: emails,
          isActive: (status.toLowerCase() === 'active')
        };
      }
    });

    return { success: true, roster: roster };
  } catch (err) {
    return { success: false, message: err.toString(), roster: {} };
  }
}

/**
 * Saves or updates department roster emails from Settings
 */
function saveNotificationRoster(rosterMap) {
  const lock = LockService.getScriptLock();
  try {
    lock.waitLock(10000);
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    let sheet = ss.getSheetByName('Notification_Roster');

    if (!sheet) {
      getNotificationRoster();
      sheet = ss.getSheetByName('Notification_Roster');
    }

    const rows = sheet.getLastRow() > 1 ? sheet.getRange(2, 1, sheet.getLastRow() - 1, 4).getValues() : [];
    const now = new Date();

    Object.entries(rosterMap).forEach(([dept, conf]) => {
      const cleanDept = dept.trim().toUpperCase();
      const emailStr = Array.isArray(conf.emails) ? conf.emails.join(', ') : String(conf.emails || '').trim();
      const status = conf.isActive ? 'Active' : 'Inactive';

      let found = false;
      for (let i = 0; i < rows.length; i++) {
        if (String(rows[i][0]).trim().toUpperCase() === cleanDept) {
          sheet.getRange(i + 2, 2, 1, 3).setValues([[emailStr, status, now]]);
          found = true;
          break;
        }
      }

      if (!found) {
        sheet.appendRow([cleanDept, emailStr, status, now]);
      }
    });

    return { success: true, message: 'Notification roster updated successfully.' };
  } catch (err) {
    return { success: false, message: err.toString() };
  } finally {
    lock.releaseLock();
  }
}

/**
 * Resolves active emails for PO staff assigned to the monitored branch
 */
function getPoStaffEmails(poName) {
  const emails = new Set();
  try {
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const usersSheet = ss.getSheetByName('Users');
    if (usersSheet && usersSheet.getLastRow() > 1) {
      const data = usersSheet.getRange(2, 1, usersSheet.getLastRow() - 1, 7).getValues();
      const targetPo = String(poName || '').trim().toLowerCase();

      data.forEach(r => {
        const userEmail = String(r[0] || '').trim().toLowerCase();
        const scope = String(r[5] || '').toLowerCase();
        const status = String(r[6] || '').toLowerCase();

        if (status === 'active' && userEmail.includes('@')) {
          if (scope.includes(targetPo) || scope.includes('all po') || scope === 'all' || scope.includes('national')) {
            emails.add(userEmail);
          }
        }
      });
    }
  } catch (e) {
    console.warn("PO Staff lookup error: " + e.toString());
  }
  return Array.from(emails);
}

/**
 * Validates and sanitizes comma/semicolon-separated email list
 */
function sanitizeEmailList(rawString) {
  if (!rawString) return [];
  return String(rawString)
    .split(/[,;\s]+/)
    .map(e => e.trim().toLowerCase())
    .filter(e => e && e.includes('@') && e.indexOf('.') > e.indexOf('@'));
}

/**
 * Dispatches findings notification to both Department Roster and Branch Staff
 */
function dispatchFindingNotification(batchPayload) {
  try {
    const poName = String(batchPayload.poMonitored || 'General Office').trim();
    const auditorName = String(batchPayload.monitoredBy || 'Monitoring Officer').trim();
    const auditDate = safeFormatReportDate(batchPayload.dateMonitoring) || safeFormatReportDate(new Date());
    const items = batchPayload.findings || [];

    if (!items.length) {
      Logger.log("dispatchFindingNotification: No findings provided. Skipping email.");
      return { success: true, dispatched: 0 };
    }

    // 1. Group findings by department
    const deptGroups = {};
    items.forEach(f => {
      let dept = 'COMPLIANCE AND RISK';
      const rawPart = String(f.particularFinding || '');
      if (rawPart.includes(' - ')) {
        dept = rawPart.split(' - ')[0].trim().toUpperCase();
      } else if (f.department) {
        dept = String(f.department).trim().toUpperCase();
      }
      if (!deptGroups[dept]) {
        deptGroups[dept] = [];
      }
      deptGroups[dept].push(f);
    });

    // 2. Fetch rosters and branch staff emails
    const rosterRes = getNotificationRoster();
    const roster = rosterRes.success ? rosterRes.roster : {};
    const poEmails = getPoStaffEmails(poName);

    // Get Web App URL for Call to Action button
    let webAppUrl = '';
    try {
      webAppUrl = ScriptApp.getService().getUrl();
    } catch (e) {
      webAppUrl = '';
    }
    if (!webAppUrl) {
      try {
        webAppUrl = SpreadsheetApp.getActiveSpreadsheet().getUrl();
      } catch (e) {
        webAppUrl = 'https://script.google.com';
      }
    }

    let dispatchCount = 0;

    // 3. Send email per department group
    Object.entries(deptGroups).forEach(([dept, deptFindings]) => {
      // Find matching roster entry (case-insensitive and partial match support)
      let deptConf = roster[dept];
      if (!deptConf) {
        // Try finding partial match in roster
        const matchKey = Object.keys(roster).find(k => k.includes(dept) || dept.includes(k));
        if (matchKey) {
          deptConf = roster[matchKey];
        }
      }

      const recipientSet = new Set();

      // Add Department Roster emails
      if (deptConf && deptConf.isActive !== false) {
        const cleanEmails = sanitizeEmailList(deptConf.emails);
        cleanEmails.forEach(e => recipientSet.add(e));
      }

      // Add Branch Staff emails
      poEmails.forEach(e => recipientSet.add(e));

      // Fallback safety: If no recipient could be resolved, alert active user
      if (recipientSet.size === 0) {
        const fallback = Session.getActiveUser().getEmail();
        if (fallback) recipientSet.add(fallback);
      }

      const validRecipients = Array.from(recipientSet).filter(e => e.includes('@'));
      if (validRecipients.length === 0) {
        Logger.log(`No valid email found for department ${dept}. Skipping.`);
        return;
      }

      const toAddresses = validRecipients.join(',');
      const subject = `[ACTION REQUIRED] Audit Findings Logged: ${dept} - PO ${poName} (${auditDate})`;

      const emailHtml = buildFindingNotificationEmailTemplate({
        poMonitored: poName,
        department: dept,
        dateMonitoring: auditDate,
        monitoredBy: auditorName,
        findings: deptFindings,
        webAppUrl: webAppUrl
      });

      // Send email using GmailApp with fallback to MailApp
      try {
        GmailApp.sendEmail(toAddresses, subject, `New audit findings logged for PO ${poName}. Please open the PMT portal to comply with Action Taken.`, {
          htmlBody: emailHtml,
          name: 'CARD MBA Monitoring Portal'
        });
        Logger.log(`Dispatched finding notification email to: ${toAddresses}`);
        dispatchCount++;
      } catch (gmailErr) {
        Logger.log("GmailApp send error, falling back to MailApp: " + gmailErr.toString());
        MailApp.sendEmail({
          to: toAddresses,
          subject: subject,
          htmlBody: emailHtml,
          name: 'CARD MBA Monitoring Portal'
        });
        dispatchCount++;
      }
    });

    return { success: true, dispatched: dispatchCount };
  } catch (err) {
    Logger.log("dispatchFindingNotification fatal error: " + err.toString());
    return { success: false, message: err.toString() };
  }
}
