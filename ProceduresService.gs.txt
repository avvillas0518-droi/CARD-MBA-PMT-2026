/**
 * CARD MBA PMT 2026 - Monitoring Procedures Guide Service
 */

function getMonitoringProceduresList() {
  try {
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    let sheet = ss.getSheetByName('Monitoring Procedures');

    // Auto-seed if sheet does not exist or only contains headers
    if (!sheet || sheet.getLastRow() < 2) {
      seedMonitoringProceduresSheet();
      sheet = ss.getSheetByName('Monitoring Procedures');
    }

    const data = sheet.getRange(2, 1, sheet.getLastRow() - 1, 6).getValues();
    const procedures = data.map((r, idx) => ({
      id: idx + 1,
      area: String(r[0] || '').trim(),
      particular: String(r[1] || '').trim(),
      documents: String(r[2] || '').trim(),
      procedure: String(r[3] || '').trim(),
      remarks: String(r[4] || '').trim(),
      classification: String(r[5] || 'Major').trim()
    })).filter(p => p.particular && p.procedure);

    return {
      success: true,
      procedures: procedures,
      areas: Array.from(new Set(procedures.map(p => p.area))).sort()
    };
  } catch (err) {
    return { success: false, message: err.toString(), procedures: [], areas: [] };
  }
}
