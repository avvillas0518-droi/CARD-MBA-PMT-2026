function setActiveSheetByName(name) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  if (!ss) return;
  const sheet = ss.getSheetByName(name);
  if (sheet) {
    ss.setActiveSheet(sheet);
  } else {
    SpreadsheetApp.getUi().alert(`Sheet "${name}" not found. Please verify workbook tabs.`);
  }
}

function exportWorkingPaperCsv() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sheet = ss.getSheetByName('Working_Paper');
  if (!sheet) {
    SpreadsheetApp.getUi().alert('Working_Paper tab does not exist.');
    return;
  }
  const rows = sheet.getDataRange().getValues();
  let csv = rows.map(r => r.map(c => `"${String(c).replace(/"/g, '""')}"`).join(',')).join('\r\n');
  const file = DriveApp.createFile(`Working_Paper_${Utilities.formatDate(new Date(), 'GMT+8', 'yyyyMMdd_HHmm')}.csv`, csv, MimeType.CSV);
  SpreadsheetApp.getUi().alert('Export created in Drive: ' + file.getUrl());
}
