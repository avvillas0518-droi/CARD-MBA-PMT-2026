function fetchSettingsCatalog() {
  try {
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const dropSheet = ss.getSheetByName('Dropdowns');
    const findSheet = ss.getSheetByName('FINDINGS');

    const categories = new Set();
    const dropdownList = [];
    if (dropSheet && dropSheet.getLastRow() > 1) {
      const data = dropSheet.getRange(2, 1, dropSheet.getLastRow() - 1, 3).getValues();
      data.forEach((r, idx) => {
        if (r[0] && r[1]) {
          categories.add(String(r[0]).trim());
          dropdownList.push({ rowId: idx + 2, type: 'dropdown', category: String(r[0]).trim(), value: String(r[1]).trim() });
        }
      });
    }

    const deptSet = new Set();
    const findingsList = [];
    if (findSheet && findSheet.getLastRow() > 1) {
      const fData = findSheet.getRange(2, 1, findSheet.getLastRow() - 1, 2).getValues();
      fData.forEach((r, idx) => {
        if (r[0] && r[1]) {
          deptSet.add(String(r[0]).trim());
          findingsList.push({ rowId: idx + 2, type: 'finding', category: String(r[0]).trim(), value: String(r[1]).trim() });
        }
      });
    }

    return {
      success: true,
      categories: Array.from(categories).sort(),
      departments: Array.from(deptSet).sort(),
      dropdowns: dropdownList,
      findings: findingsList,
      spreadsheetUrl: ss.getUrl()
    };
  } catch (err) {
    return { success: false, message: err.toString() };
  }
}

function addParameterEntry(type, category, value) {
  try {
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const sheetName = type === 'finding' ? 'FINDINGS' : 'Dropdowns';
    const sheet = ss.getSheetByName(sheetName);
    if (!sheet) throw new Error(`${sheetName} sheet not found.`);
    sheet.appendRow([category.trim(), value.trim(), 'Active']);
    return { success: true, message: `Added "${value}" to [${category}].` };
  } catch (err) {
    return { success: false, message: err.toString() };
  }
}

function removeParameterEntry(type, category, value) {
  try {
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const sheetName = type === 'finding' ? 'FINDINGS' : 'Dropdowns';
    const sheet = ss.getSheetByName(sheetName);
    if (!sheet) throw new Error(`${sheetName} not found.`);
    const rows = sheet.getDataRange().getValues();
    for (let i = rows.length - 1; i >= 1; i--) {
      if (String(rows[i][0]).trim() === category && String(rows[i][1]).trim() === value) {
        sheet.deleteRow(i + 1);
        return { success: true, message: `Removed "${value}".` };
      }
    }
    return { success: false, message: 'Item not found.' };
  } catch (err) {
    return { success: false, message: err.toString() };
  }
}

function unlockMasterSheets() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  ['FINDINGS', 'Dropdowns', 'Office_Master', 'Monitoring Procedures'].forEach(name => {
    const sheet = ss.getSheetByName(name);
    if (sheet) {
      sheet.getProtections(SpreadsheetApp.ProtectionType.SHEET).forEach(p => p.remove());
    }
  });
  return { success: true, message: 'Master sheets unlocked.' };
}

function protectMasterSheets() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const user = Session.getEffectiveUser().getEmail();
  ['FINDINGS', 'Dropdowns', 'Office_Master', 'Monitoring Procedures'].forEach(name => {
    const sheet = ss.getSheetByName(name);
    if (sheet) {
      const p = sheet.protect().setDescription(`Protected: ${name}`);
      p.removeEditors(p.getEditors());
      p.addEditor(user);
    }
  });
  return { success: true, message: 'Master sheet protections enforced.' };
}
